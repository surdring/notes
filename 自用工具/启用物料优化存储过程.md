```sql
CREATE OR REPLACE procedure proc_enable_material(enable_date in varchar2) as
BEGIN
  -- 启用物料（视图表）
  merge into BD_MATERIAL_V v
  using (
    select DISTINCT code
    from pre_enable_material
    where is_en = '0' and rq = enable_date
  ) t
  on (t.code = v.code)
  when matched then
    update set v.enablestate = 2,
               v.memo = '启用日期:' || to_char(sysdate,'yyyy-mm-dd hh24:mi:ss');

  -- 启用物料（主表）
  merge into BD_MATERIAL m
  using (
    select DISTINCT code
    from pre_enable_material
    where is_en = '0' and rq = enable_date
  ) t
  on (t.code = m.code)
  when matched then
    update set m.enablestate = 2,
               m.memo = '启用日期:' || to_char(sysdate,'yyyy-mm-dd hh24:mi:ss');

  -- 先标记：存在的编码 => 处理成功 is_en='1'
  merge into pre_enable_material pem
  using (
    select DISTINCT pem.rowid rid
    from pre_enable_material pem
    join bd_material m on m.code = pem.code
    where pem.is_en='0' and pem.rq = enable_date
  ) t
  on (pem.rowid = t.rid)
  when matched then
    update set pem.is_en='1';

  -- 再标记：剩余未处理的 => 不存在 is_en='2'
  update pre_enable_material
  set is_en='2'
  where is_en='0' and rq = enable_date;

END;
```
