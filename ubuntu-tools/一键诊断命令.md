bash -lc '
out="$HOME/diag_$(hostname)_$(date +%F_%H%M%S).log"
{
  echo "=== TIME ==="; date
  echo "=== HOST ==="; hostname; uname -a
  echo
  echo "=== UPTIME/LOAD ==="; uptime
  echo
  echo "=== CPU TOP (by %CPU) ==="
  ps -eo pid,ppid,comm,%cpu,%mem,stat --sort=-%cpu | head -n 25
  echo
  echo "=== MEM ==="; free -h
  echo
  echo "=== DISK SPACE ==="; df -h
  echo
  echo "=== DISK IO (iostat -xz 1 3) ==="
  if command -v iostat >/dev/null 2>&1; then
    iostat -xz 1 3
  else
    echo "iostat not installed (install: sudo apt install sysstat)"
  fi
  echo
  echo "=== VMSTAT (1s x 5) ==="
  if command -v vmstat >/dev/null 2>&1; then
    vmstat 1 5
  else
    echo "vmstat not installed"
  fi
  echo
  echo "=== GPU (ROCm SMI) ==="
  if command -v rocm-smi >/dev/null 2>&1; then
    rocm-smi --alldevices --showtemp --showpower --showclocks --showperflevel --showmemuse --showpids
  else
    echo "rocm-smi not found"
  fi
  echo
  echo "=== KERNEL LOG (last 200 lines, filtered) ==="
  sudo dmesg -T | tail -n 200 | grep -Ei "oom|killed process|out of memory|amdgpu|kfd|reset|gpu fault|timeout" || true
} | tee "$out"
echo
echo "Report saved to: $out"
'
